\begin{align}
 F_\perp &= \FskF = \mKi\cdot\gE\cdot\cos\left(\wiO\right) = \Fsk \approx \result{\FskP{0}{2}} \\
 F_\parallel &= \FpaF = \mKi\cdot\gE\cdot\sin\left(\wiO\right) = \Fpa \approx \result{\FpaP{0}{2}}
\end{align}
