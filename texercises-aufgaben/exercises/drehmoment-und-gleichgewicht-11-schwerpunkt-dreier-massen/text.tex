 \fill (2.4,0) circle (0.22) node[above=6pt] {$m_2$};
 \fill (8.4,0) circle (0.22) node[above=6pt] {$m_3$};
 \draw[|<->|] (0,-0.6) -- (2.4,-0.6) node[midway, below] {$a$};
 \draw[|<->|] (2.4,-0.6) -- (8.4,-0.6) node[midway, below] {$b$};
