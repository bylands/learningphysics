Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc $\aLO = \aL \;\result{>}\; \aRO$
\abc $\bLO = \bL \;\result{<}\; \bRO$
\abc $\cLO = \cLP{3}{2} \;\result{<}\; \cRO$
\abc $\dPO + \dQO = \dLP{-3}{3} \;\result{<}\; \dRO$
\abc $\eAO\cdot\left(\eBO\right)^2 = \eA\cdot\left(\eB\right)^2 = \eLP{-3}{2} \;\result{<}\; \eRO$
\abc $\fRO = \fRP{-9}{2}$, also $\fLO \;\result{=}\; \fRO$
\end{abcliste}
