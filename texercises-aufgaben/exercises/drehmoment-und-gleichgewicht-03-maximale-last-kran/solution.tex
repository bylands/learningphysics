\begin{align}
 m_\mathrm{L}\,g\cdot a_\mathrm{L} &= m_\mathrm{G}\,g\cdot a_\mathrm{G} \\
 m_\mathrm{L} &= \mLaF = \mKr\cdot\frac{\bKr}{\aKr} = \mLa \approx \result{\mLatP{0}{2}}
\end{align}
