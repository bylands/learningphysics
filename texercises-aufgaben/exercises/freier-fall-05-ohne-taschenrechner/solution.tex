Für das Kopfrechnen genügt $g \approx \gzO$; der Fehler ist dann kleiner als \SI{2}{\percent}. Zum Vergleich ist jeweils auch der Wert mit $g = \gEO$ angegeben.
\begin{abcliste}
\abc
\begin{align}
 v &= \vaF \approx \gz\cdot\ta = \vaz \\
 v &= \gE\cdot\ta = \va \approx \result{\vaP{0}{2}}
\end{align}

\abc
\begin{align}
 h &= \hbF \approx \frac{1}{2}\cdot\gz\cdot\left(\tb\right)^2 = \hbz \\
 h &= \frac{1}{2}\cdot\gE\cdot\left(\tb\right)^2 = \hb \approx \result{\hbP{0}{2}}
\end{align}

\abc
\begin{align}
 t &= \tcF \approx \sqrt{\frac{2\cdot\hc}{\gz}} = \sqrt{\tqzP{0}{1}} \approx \tczP{0}{2} \\
 t &= \sqrt{\frac{2\cdot\hc}{\gE}} = \tc \approx \result{\tcP{0}{2}}
\end{align}
\end{abcliste}
