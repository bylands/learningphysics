Für das Kopfrechnen genügt $g \approx \SI{10}{\meter\per\second\squared}$; der Fehler ist dann kleiner als \SI{2}{\percent}. In Klammern stehen die Werte mit $g = \gEO$.
\begin{abcliste}
\abc $v = g\,t \approx \SI{10}{\meter\per\second\squared}\cdot\SI{5}{\second} = \SI{50}{\meter\per\second}$ \quad ($\vaF = \va \approx \result{\vaP{0}{2}}$)
\abc $h = \frac{1}{2}g\,t^2 \approx \SI{5}{\meter\per\second\squared}\cdot\SI{16}{\second\squared} = \SI{80}{\meter}$ \quad ($\hb \approx \result{\hbP{0}{2}}$)
\abc $t = \sqrt{2h/g} \approx \sqrt{\SI{6}{\second\squared}} \approx \SI{2.5}{\second}$ \quad ($\tcF = \tc \approx \result{\tcP{0}{2}}$)
\end{abcliste}
