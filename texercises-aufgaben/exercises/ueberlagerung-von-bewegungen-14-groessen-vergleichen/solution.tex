Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc $\aAO\cdot\aBO = \aLP{0}{2} \;\result{=}\; \aRO$
\abc $\bLO = \bLsP{0}{2} \;\result{=}\; \bRO$
\abc $\cRO = \cRsP{-6}{2}$, also $\cLO \;\result{=}\; \cRO$
\abc $\dLO = \dLsP{0}{2} \;\result{=}\; \dRsP{0}{2} = \dRO$
\abc Eine Länge und eine Zeit lassen sich nicht subtrahieren; $\eAO - \eBO$ ist sinnlos: $\result{\neq}$
\abc $\fLO = \fLkP{0}{2} \;\result{>}\; \fRO$
\end{abcliste}
