Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc $\aLO = \aLkP{0}{2} \;\result{<}\; \aRO$
\abc Eine Fläche lässt sich nicht mit einem Volumen vergleichen: $\bLO \;\result{\neq}\; \bRO$
\abc $\cLO = \cLsP{-6}{2} \;\result{=}\; \cRO$
\abc $\dLO \;\result{<}\; \dRO = \dRsP{-9}{2}$
\abc Meter und Sekunden lassen sich nicht addieren, beide Ausdrücke sind sinnlos: $\result{\neq}$
\abc $10^{-5} : 10^{-9} = 10^{-5-(-9)} = 10^{4} \;\result{>}\; 10^{-14}$
\end{abcliste}
