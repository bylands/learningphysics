\begin{abcliste}
\abc Die Tabelle \emph{Geophysikalische Daten der Schweiz} (FoTaBe S.~204) gibt für Olten (\ang{47;21.1;} N, \SI{401}{\meter} ü.\,M.)
\begin{align}
 g &= \gol \approx \result{\golP{0}{4}}
\end{align}

\abc Mit der Masse \MMO\ und dem Äquatorradius \rMO\ des Mars (FoTaBe S.~212) gilt
\begin{align}
 g_\mathrm{M} &= \gmaF \\
   &= \frac{\ncG\cdot\MM}{\left(\rM\right)^2} \\
   &= \gma \approx \result{\gmaP{0}{2}}
\end{align}
Das stimmt mit der Schwerebeschleunigung in der Planetentabelle überein.

\abc Für einen Ort der geografischen Breite $\varphi = \brO$ und der Höhe $h = \hevO$ über Meer gilt näherungsweise (FoTaBe S.~209) mit $k_1 = \kaO$, $k_2 = \kbO$ und $g_\text{Ä} = \gaeqO$:
\begin{align}
 g &= \gevF \\
   &= \left(1+\ka\cdot\sin^2\left(\brO\right)+\left(\kb\right)\cdot\hev\right)\cdot\gaeq \\
   &= \gev \approx \result{\gevP{0}{3}}
\end{align}
\end{abcliste}
