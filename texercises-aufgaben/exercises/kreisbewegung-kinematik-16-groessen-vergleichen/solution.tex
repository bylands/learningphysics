Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc $\aLO = \aLsP{0}{2} \;\result{>}\; \aRsP{0}{2} = \aRO$
\abc $\bLO = \bLrP{0}{4} \;\result{=}\; \bRP{0}{4}$, d.h. $\bLO$ entspricht im Bogenmass $3\pi/2$.
\abc $\cLO = \cLsP{-12}{2} \;\result{<}\; \cRsP{-12}{3} = \cRO$
\abc $\dLO = \dLsP{0}{2} \;\result{=}\; \dRO$ (Umdrehungen pro Sekunde)
\abc $\eLO = \eLdP{0}{3} \;\result{<}\; \eRO$
\abc $\fAO\cdot\fBO = \fA\cdot\fB = \fL \;\result{<}\; \fRsP{0}{2} = \fRO$
\end{abcliste}
