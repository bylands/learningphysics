Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc $\aLO \;\result{>}\; \aRO = \aRkP{0}{2}$
\abc $\bLO = \bLrP{0}{3} \;\result{=}\; 3\pi/2 = \bRP{0}{3}$ (exakt gleich, da $\ang{360}$ dem Winkel $2\pi$ entspricht)
\abc $\cLO \;\result{<}\; \cRO = \cRsP{-12}{2}$
\abc $\dLO = \dLsP{0}{2} \;\result{=}\; \dRO$ (Umdrehungen pro Minute in Umdrehungen pro Sekunde)
\abc $\eLO = \eLdP{0}{3} \;\result{<}\; \eRO$
\abc $\fAO\cdot\fBO\cdot\left(\fCO\right)^2 = \fLP{0}{2} \;\result{>}\; \fRO$, denn $\si{\kilo\gram\meter\per\second\squared} = \si{\newton}$
\end{abcliste}
