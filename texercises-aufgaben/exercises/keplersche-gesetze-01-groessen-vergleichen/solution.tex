Man rechnet jeweils beide Grössen in dieselbe Einheit um ($\SI{1}{\AE} = \AEmO$, FoTaBe S.~205).
\begin{abcliste}
\abc $\aLiO = \aLi \;\result{>}\; \aReO$
\abc $\bLiO = \bLi \;\result{=}\; \bRe = \bReO$
\abc $\cLiO$ ist ein Volumen, $\cReO$ eine Masse: $\cLiO \;\result{\neq}\; \cReO$
\abc $\dReO = \dRe$, also $\dLiO \;\result{=}\; \dReO$
\abc $\eLiO = \eLi$, also $\eLiO \;\result{=}\; \eReO$
\abc $\fAO\cdot\fBO = \fA\cdot\fB = \fLiP{3}{2} \;\result{<}\; \fReO$
\end{abcliste}
