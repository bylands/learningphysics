Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc $\aLO \;\result{\neq}\; \aRO$: Eine Masse lässt sich nicht mit einem Volumen vergleichen.
\abc $\bLO = \bLsP{0}{2} \;\result{=}\; \bRO$
\abc $\cAO\cdot\cBO\cdot\left(\cCO\right)^2 = \cLP{0}{2} \;\result{>}\; \cRO$, denn $\si{\kilo\gram\meter\per\second\squared} = \si{\newton}$
\abc $\dLO = \dLsP{-9}{2} \;\result{=}\; \dRO$
\abc $\eLO \;\result{>}\; \eRO = \eRsP{0}{2}$
\abc $\fLO = \fLkP{0}{2} \;\result{>}\; \fRO$
\end{abcliste}
