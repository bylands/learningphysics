Man rechnet jeweils beide Grössen in dieselbe Einheit um.
\begin{abcliste}
\abc Kraft und Masse sind verschiedene Grössen: $\aLO \;\result{\neq}\; \aRO$
\abc $\bLO = \bL \;\result{=}\; \bRO$
\abc $\cMO\cdot\cAO = \cLP{0}{3} \;\result{<}\; \cRO$
\abc Wegen $\si{\newton} = \si{\kilo\gram\meter\per\second\squared}$ ist $\si{\newton\second} = \si{\kilo\gram\meter\per\second}$, also $\dPO = \dPsP{0}{2} \;\result{=}\; \dRO$
\abc $\eLO = \eLsP{-3}{2} \;\result{>}\; \eRO$
\abc $\fLO = \fL \;\result{<}\; \fR = \fRO$
\end{abcliste}
