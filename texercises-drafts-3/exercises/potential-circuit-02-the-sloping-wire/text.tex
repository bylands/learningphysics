A \SI{10}{\V} battery drives a current $I$ through two lamps $L_1$ and $L_2$. The voltage across $L_1$ is \SI{4}{\V}, across $L_2$ \SI{6}{\V}. The potential is measured from the negative terminal of the battery at A: $\varphi_A = \SI{0}{\V}$.

\begin{center}

 \draw (0,3.2) to[short, i={$I$}] (3.0,3.2);

 \fill (0,0) circle (2pt); \node[below left] at (0,0) {A};

\end{circuitikz}
\end{center}
